        // Создаем карту и центрируем её
        var map = L.map('map').setView([0, 0], 16);

        // Добавляем тайлы (путь '{z}/{x}/{y}.png' берет папки 16, 17, 18 и т.д. из корня сайта)
        L.tileLayer('{z}/{x}/{y}.png', {
            minZoom: 15,
            maxZoom: 20,
            tms: false,
            attribution: 'Ортофотоплан'
        }).addTo(map);
